async function translateText() { const text = inputText.value.trim(); const from = document.getElementById("fromLang").value; const to = document.getElementById("toLang").value; if (!text) { alert("Skriv en text eller ta kort på ett dokument först."); return; } if (from === to) { result.innerText = text; statusText.innerText = "✅ Texten är redan på valt språk"; return; } statusText.innerText = "🌍 Översätter..."; result.innerText = "Översätter..."; try { const chunks = splitText(text, 450); let translatedText = ""; for (const chunk of chunks) { const url = "https://api.mymemory.translated.net/get?q=" + encodeURIComponent(chunk) + "&langpair=" + encodeURIComponent(from + "|" + to); console.log("Translation URL:", url); const response = await fetch(url); console.log("HTTP status:", response.status); if (!response.ok) { throw new Error( "HTTP-fel från översättningstjänsten: " + response.status ); } const data = await response.json(); console.log("Translation response:", data); if ( !data.responseData || !data.responseData.translatedText ) { throw new Error( "Översättning saknas i API-svaret" ); } translatedText += data.responseData.translatedText + "\n"; } result.innerText = translatedText.trim(); statusText.innerText = "✅ Översättning klar"; } catch (error) { console.error("TRANSLATION ERROR:", error); result.innerText = "Översättningen kunde inte genomföras.\n\n" + error.message; statusText.innerText = "❌ Översättningen misslyckades"; } }
